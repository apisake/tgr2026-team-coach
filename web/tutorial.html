3. อัปโหลดสเก็ตช์ตรวจบอร์ด

เปิด L00_board_check/L00_board_check.ino อัปโหลด แล้วเปิด Serial Monitor ที่ 115200 baud (ถ้าไม่เห็นข้อความแรก ให้กด RESET บนบอร์ดหลังเปิด Serial Monitor)

สเก็ตช์จะทำตามลำดับ:

  1. LED GPIO18 ติด 1 วินาที — ถ้าไม่ติดแต่ดับแทน แสดงว่า LED ต่อแบบ active-low ให้จดไว้
  2. เสียงที่ 1 ออกจาก GPIO26 แล้ว เสียงที่ 2 ออกจาก GPIO25 — จดว่าลำโพงบนบอร์ดดังที่เสียงไหน
  3. สแกน I2C — ควรเจอ OLED ที่ 0x3C (หรือ 0x3D)
  4. วนแสดง ค่าแรงดันของขา GPIO36 และสถานะปุ่ม — หมุน KNOB บนบอร์ดแล้วค่าควรเปลี่ยน, กด BUTTON แล้ว LED ควรติด

ถ้าผลต่างจากที่คาดไว้ ให้แก้ BUZZER_PIN, OUT_PIN, KNOB_PIN หรือ OLED ที่หัวไฟล์ของสเก็ตช์บทถัดไป และแก้ตารางใน README ด้วย

ข้อควรระวังของ ESP32

  • ลอจิก 3.3 V — ห้ามป้อน 5 V เข้าขาสัญญาณ
  • Strapping pins 0, 2, 5, 12, 15 กำหนดโหมดตอนบูต: อย่าต่ออุปกรณ์ที่ดึงขาเหล่านี้ HIGH/LOW ค้างไว้ตอนรีเซ็ต (โดยเฉพาะ GPIO12 ต้องเป็น LOW ตอนบูต)
  • GPIO34–39 เป็น input อย่างเดียวและไม่มี pull-up/pull-down ภายใน
  • ADC2 (GPIO0, 2, 4, 12–15, 25–27) ใช้อ่าน analog ไม่ได้ขณะเปิด Wi-Fi — บอร์ดนี้ให้พอร์ต analog เป็น ADC1 (GPIO32–36) อยู่แล้ว

// L00 · ตรวจบอร์ด IPST-WiFi: LED, ลำโพง, I2C, ปุ่มหมุน, ปุ่มกด
// เปิด Serial Monitor ที่ 115200 แล้วจดผลตามตารางใน 00-setup.md

#include <Wire.h>

const int LED_PIN = 18;
const int BUTTON_PIN = 0;   // กด = LOW
const int KNOB_PIN = 36;    // VP
const int SDA_PIN = 21;
const int SCL_PIN = 22;
const int BUZZER_CANDIDATES[] = {26, 25};

void beep(int pin, int freq, int ms) {
  ledcAttach(pin, freq, 8);
  ledcWrite(pin, 128);  // duty 50%
  delay(ms);
  ledcWrite(pin, 0);
  ledcDetach(pin);
}

void scanI2C() {
  int found = 0;
  for (uint8_t addr = 1; addr < 127; addr++) {
    Wire.beginTransmission(addr);
    if (Wire.endTransmission() == 0) {
      Serial.printf("  พบอุปกรณ์ที่ 0x%02X\n", addr);
      found++;
    }
  }
  if (found == 0) Serial.println("  ไม่พบอุปกรณ์ I2C");
}

void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n=== IPST-WiFi board check ===");

  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT);  // มี pull-up บนบอร์ดแล้ว

  Serial.println("[1] LED GPIO18 = HIGH 1 วินาที (ติดหรือไม่?)");
  digitalWrite(LED_PIN, HIGH);
  delay(1000);
  digitalWrite(LED_PIN, LOW);

  for (int i = 0; i < 2; i++) {
    int pin = BUZZER_CANDIDATES[i];
    Serial.printf("[2.%d] เสียงที่ %d จาก GPIO%d\n", i + 1, i + 1, pin);
    beep(pin, 2000, 400);
    delay(800);
  }

  Serial.println("[3] สแกน I2C (SDA 21, SCL 22)");
  Wire.begin(SDA_PIN, SCL_PIN);
  scanI2C();

  Serial.println("[4] หมุน KNOB และกด BUTTON — ค่าจะแสดงทุก 0.3 วินาที");
}

void loop() {
  int mv = analogReadMilliVolts(KNOB_PIN);
  bool pressed = digitalRead(BUTTON_PIN) == LOW;
  digitalWrite(LED_PIN, pressed ? HIGH : LOW);
  Serial.printf("knob(GPIO%d) = %4d mV   button = %s\n", KNOB_PIN, mv, pressed ? "กด" : "-");
  delay(300);
}

1GPIO: LED, ปุ่ม และ millis()

เรียนอะไร

  • pinMode(), digitalWrite(), digitalRead()
  • ทำไมไม่ใช้ delay(): ระหว่าง delay(500) ESP32 ทำอย่างอื่นไม่ได้เลย ทั้งอ่านปุ่ม อ่านเซ็นเซอร์ และรักษาการเชื่อมต่อ MQTT ในบทหลัง ๆ จึงใช้รูปแบบนี้แทน
    C++
    if (millis() - lastTime >= INTERVAL) {
  lastTime = millis();
  // ทำงานตามรอบ
}

    การลบแบบ millis() - lastTime ยังถูกต้องแม้ millis() วนกลับเป็น 0 (ทุก ~49 วัน)

  • Debounce: หน้าสัมผัสปุ่มเด้งหลายครั้งในไม่กี่มิลลิวินาที จึงต้องรอให้ค่านิ่ง (30 ms) ก่อนเชื่อ
  • Edge detection: สนใจ "จังหวะที่เปลี่ยนเป็นกด" ไม่ใช่ "กำลังกดอยู่" — ฟังก์ชัน buttonPressed() คืน true ครั้งเดียวต่อการกด

ฮาร์ดแวร์

ใช้ของบนบอร์ดอย่างเดียว: LED GPIO18 และปุ่ม BUTTON GPIO0 (GPIO0 มีตัวต้านทาน pull-up บนบอร์ดแล้ว จึงใช้ INPUT ได้ ถ้าใช้ปุ่มภายนอกบนพอร์ต K3 GPIO5 ให้ใช้ INPUT_PULLUP)

ลองทำ

อัปโหลด L01_gpio.ino แล้วกด BUTTON — LED จะสลับระหว่างกระพริบช้า เร็ว และดับ

แบบฝึกหัด

  1. เพิ่มโหมดที่ 4 "ติดค้าง"
  2. กดค้าง 2 วินาที ให้กลับไปโหมดแรก (ใบ้: จำเวลาที่เริ่มกดไว้)
  3. ต่อ ZX-SWITCH01 ที่พอร์ต K3 (GPIO5) และ ZX-LED ที่ K5 (GPIO23) ให้ปุ่มภายนอกควบคุม LED ภายนอกแยกจากปุ่มบนบอร์ด — ทั้งสองคู่ต้องทำงานพร้อมกันได้โดยไม่รอกัน
  4. คิดต่อกับโจทย์: ถ้าเกษตรกรกดปุ่มเพื่อ "เริ่มล็อตใหม่" ควรใช้กดสั้นหรือกดค้าง เพราะอะไร

// L01 · GPIO: LED กระพริบแบบไม่บล็อก + ปุ่มกดพร้อม debounce
// กด BUTTON เพื่อสลับโหมด: กระพริบช้า → กระพริบเร็ว → ดับ

const int LED_PIN = 18;
const int BUTTON_PIN = 0;  // กด = LOW (pull-up บนบอร์ด)

const unsigned long DEBOUNCE_MS = 30;
const unsigned long BLINK_MS[] = {500, 100, 0};  // 0 = ดับ
const char *MODE_NAME[] = {"ช้า", "เร็ว", "ดับ"};

int mode = 0;
bool ledOn = false;
unsigned long lastBlink = 0;

// สถานะปุ่มสำหรับ debounce
bool stableState = HIGH;
bool lastReading = HIGH;
unsigned long lastChange = 0;

// คืนค่า true ครั้งเดียวต่อการกดหนึ่งครั้ง
bool buttonPressed() {
  bool reading = digitalRead(BUTTON_PIN);
  if (reading != lastReading) {
    lastChange = millis();
    lastReading = reading;
  }
  if (millis() - lastChange > DEBOUNCE_MS && reading != stableState) {
    stableState = reading;
    return stableState == LOW;
  }
  return false;
}

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT);
  Serial.println("L01 GPIO — กด BUTTON เพื่อเปลี่ยนโหมด");
}

void loop() {
  if (buttonPressed()) {
    mode = (mode + 1) % 3;
    Serial.printf("โหมด: %s\n", MODE_NAME[mode]);
  }

  unsigned long now = millis();
  if (BLINK_MS[mode] == 0) {
    ledOn = false;
  } else if (now - lastBlink >= BLINK_MS[mode]) {
    lastBlink = now;
    ledOn = !ledOn;
  }
  digitalWrite(LED_PIN, ledOn ? HIGH : LOW);
}

2PWM (LEDC): หรี่ไฟและเสียงแจ้งเตือน

เรียนอะไร

  • PWM คือการเปิด-ปิดขาเร็ว ๆ แล้วปรับสัดส่วนเวลาที่เปิด (duty cycle) — ใช้ปรับความสว่าง LED, ความเร็วพัดลมผ่าน MOSFET, มุม servo และสร้างเสียง
  • ESP32 มีวงจร PWM ชื่อ LEDC ใน core 3.x ใช้แค่ 3 ฟังก์ชัน:
    ฟังก์ชันใช้ทำอะไร
    ledcAttach(pin, freq, bits)ผูกขากับ PWM เช่น 5000 Hz, 8 บิต (duty 0–255)
    ledcWrite(pin, duty)ตั้ง duty
    ledcWriteTone(pin, freq)เล่นเสียงความถี่ freq (0 = เงียบ)
    ตัวอย่างบนอินเทอร์เน็ตจำนวนมากยังใช้ API ของ core 2.x (ledcSetup + ledcAttachPin + เลข channel) ซึ่งคอมไพล์ไม่ผ่านบน core 3.x
  • ตัวเล่นเสียงแบบไม่บล็อก: เก็บ "กำลังเล่นโน้ตที่เท่าไร ดังอยู่หรือเงียบอยู่ เริ่มเมื่อไร" แล้วให้ loop() เรียก updateAlarm() ทุกรอบ — ไฟยังหรี่ได้ต่อเนื่องระหว่างมีเสียง

// L02 · PWM ด้วย LEDC: หรี่ไฟ LED และเสียงแจ้งเตือนจาก piezo
// ใช้ API ของ ESP32 Arduino core 3.x: ledcAttach / ledcWrite / ledcWriteTone

const int LED_PIN = 18;
const int EXT_LED_PIN = 19;  // ZX-LED ที่พอร์ต K4 (ถ้ามี)
const int BUZZER_PIN = 26;   // ยืนยันด้วย L00 ก่อน
const int BUTTON_PIN = 0;

const int PWM_FREQ = 5000;  // Hz
const int PWM_BITS = 8;     // duty 0–255

// รูปแบบเสียง: ความถี่ (Hz), เวลาดัง (ms), เวลาเงียบ (ms)
struct Tone { int freq; int onMs; int offMs; };
const Tone ALARM[] = {{2500, 150, 100}, {2500, 150, 100}, {2500, 150, 600}};
const int ALARM_LEN = sizeof(ALARM) / sizeof(ALARM[0]);

// ตัวเล่นเสียงแบบไม่บล็อก
int toneIndex = -1;  // -1 = ไม่ได้เล่น
bool toneOn = false;
unsigned long toneSince = 0;

void startAlarm() {
  toneIndex = 0;
  toneOn = true;
  toneSince = millis();
  ledcWriteTone(BUZZER_PIN, ALARM[0].freq);
}

void updateAlarm() {
  if (toneIndex < 0) return;
  const Tone &t = ALARM[toneIndex];
  unsigned long elapsed = millis() - toneSince;
  if (toneOn && elapsed >= (unsigned long)t.onMs) {
    ledcWriteTone(BUZZER_PIN, 0);  // ความถี่ 0 = เงียบ
    toneOn = false;
    toneSince = millis();
  } else if (!toneOn && elapsed >= (unsigned long)t.offMs) {
    toneIndex++;
    if (toneIndex >= ALARM_LEN) {
      toneIndex = -1;
      return;
    }
    ledcWriteTone(BUZZER_PIN, ALARM[toneIndex].freq);
    toneOn = true;
    toneSince = millis();
  }
}

// หรี่ไฟขึ้นลงแบบสามเหลี่ยม คาบ 2 วินาที
int fadeDuty() {
  unsigned long phase = millis() % 2000;
  int d = phase < 1000 ? phase : 2000 - phase;  // 0..1000..0
  return map(d, 0, 1000, 0, 255);
}

void setup() {
  Serial.begin(115200);
  pinMode(BUTTON_PIN, INPUT);
  ledcAttach(LED_PIN, PWM_FREQ, PWM_BITS);
  ledcAttach(EXT_LED_PIN, PWM_FREQ, PWM_BITS);
  ledcAttach(BUZZER_PIN, 2000, PWM_BITS);
  Serial.println("L02 PWM — กด BUTTON เพื่อเล่นเสียงแจ้งเตือน");
}

void loop() {
  int duty = fadeDuty();
  ledcWrite(LED_PIN, duty);
  ledcWrite(EXT_LED_PIN, 255 - duty);

  static bool lastBtn = HIGH;
  bool btn = digitalRead(BUTTON_PIN);
  if (lastBtn == HIGH && btn == LOW && toneIndex < 0) startAlarm();
  lastBtn = btn;

  updateAlarm();
}

3ADC: อ่านแรงดันให้แม่นและนิ่ง

เรียนอะไร

  • ESP32 มี ADC 12 บิต (0–4095) สองชุด: ADC1 (GPIO32–39) และ ADC2 — ADC2 ใช้ไม่ได้ขณะเปิด Wi-Fi บอร์ด IPST-WiFi ให้พอร์ต analog K7–K10 (GPIO32–35) และ KNOB (GPIO36) อยู่บน ADC1 ทั้งหมด
  • Attenuation กำหนดช่วงแรงดันที่วัดได้ ค่าเริ่มต้น ADC_11db วัดได้ราว 0.15–3.1 V และไม่เป็นเชิงเส้นที่ปลายทั้งสองข้าง
  • analogRead() คืนค่าดิบซึ่งแต่ละชิปเพี้ยนต่างกัน ส่วน analogReadMilliVolts() ใช้ค่า calibration ที่โรงงานเขียนไว้ใน eFuse จึงได้ mV ที่ใกล้ความจริงกว่า — ใช้ตัวนี้เสมอเมื่อต้องแปลงเป็นหน่วยจริง
  • ADC ของ ESP32 มี noise ราว ±10–30 mV จึงต้องกรอง:
    • Median filter ตัดค่ากระโดด (spike) ทิ้ง — เรียงค่า 9 ค่าล่าสุดแล้วเอาตัวกลาง
    • EMA (exponential moving average) ema = α·x + (1−α)·ema ทำให้เส้นเรียบ ใช้หน่วยความจำค่าเดียว
  • Serial Plotter ของ Arduino IDE 2.x อ่านรูปแบบ label:value,label:value และวาดกราฟหลายเส้นได้

ฮาร์ดแวร์

  • KNOB บนบอร์ด (GPIO36)
  • ZX-POT ที่พอร์ต K7 (GPIO32) ถ้ามี

// L03 · ADC: อ่านแรงดันจาก KNOB และ ZX-POT แล้วกรองสัญญาณ
// เปิด Tools → Serial Plotter (115200) เพื่อดูกราฟ raw / median / ema

const int KNOB_PIN = 36;  // KNOB บนบอร์ด (VP)
const int POT_PIN = 32;   // ZX-POT ที่พอร์ต K7

const int SAMPLE_MS = 20;    // อ่านทุก 20 ms (50 ครั้ง/วินาที)
const int MEDIAN_N = 9;      // จำนวนค่าใน median filter (เลขคี่)
const float EMA_ALPHA = 0.1; // 0..1 ยิ่งน้อยยิ่งเรียบแต่ตอบสนองช้า

int window[MEDIAN_N];
int windowPos = 0;
float ema = -1;

int median(const int *data, int n) {
  int sorted[MEDIAN_N];
  memcpy(sorted, data, n * sizeof(int));
  // insertion sort พอสำหรับ n เล็ก
  for (int i = 1; i < n; i++) {
    int v = sorted[i], j = i - 1;
    while (j >= 0 && sorted[j] > v) { sorted[j + 1] = sorted[j]; j--; }
    sorted[j + 1] = v;
  }
  return sorted[n / 2];
}

void setup() {
  Serial.begin(115200);
  analogReadResolution(12);                     // 0–4095
  analogSetPinAttenuation(KNOB_PIN, ADC_11db);  // ช่วงวัดประมาณ 0–3.1 V
  analogSetPinAttenuation(POT_PIN, ADC_11db);
  for (int i = 0; i < MEDIAN_N; i++) window[i] = analogReadMilliVolts(KNOB_PIN);
}

void loop() {
  static unsigned long last = 0;
  if (millis() - last < SAMPLE_MS) return;
  last = millis();

  int raw = analogRead(KNOB_PIN);          // ค่าดิบ 0–4095 (ไม่ได้ calibrate)
  int mv = analogReadMilliVolts(KNOB_PIN); // mV ที่ผ่าน calibration จาก eFuse

  window[windowPos] = mv;
  windowPos = (windowPos + 1) % MEDIAN_N;
  int med = median(window, MEDIAN_N);

  ema = (ema < 0) ? med : EMA_ALPHA * med + (1 - EMA_ALPHA) * ema;

  int potMv = analogReadMilliVolts(POT_PIN);

  // รูปแบบ label:value ที่ Serial Plotter ของ IDE 2.x อ่านได้
  Serial.printf("raw_x0.8:%d,mv:%d,median:%d,ema:%.0f,pot_mv:%d\n",
                (int)(raw * 0.8), mv, med, ema, potMv);
}

เรียนอะไร

  • การต่อบอร์ด ZX ผ่านพอร์ต JST 3 ขา (S, +, GND) — เสียบได้ทางเดียว ไม่ต้องใช้ breadboard

int soilDryMv = 300;
int soilWetMv = 2500;

// อ่าน n ครั้งแล้วคืน median ของค่า mV
int readMv(int pin, int n = 9) {
  int v[15];
  n = constrain(n, 1, 15);
  for (int i = 0; i < n; i++) v[i] = analogReadMilliVolts(pin);
  for (int i = 1; i < n; i++) {
    int x = v[i], j = i - 1;
    while (j >= 0 && v[j] > x) { v[j + 1] = v[j]; j--; }
    v[j + 1] = x;
  }
  return v[n / 2];
}

// MCP9701: Vout = 400 mV + 19.5 mV/°C
float mcp9701C(int mv) {
  return (mv - 400) / 19.5;
}

// แปลง mV เป็น % ตามจุด calibrate (ใช้ได้ทั้งกรณีเปียกแล้วแรงดันขึ้นหรือลง)
float soilPercent(int mv) {
  float pct = 100.0 * (mv - soilDryMv) / (soilWetMv - soilDryMv);
  return constrain(pct, 0, 100);
}

  Serial.begin(115200);
  pinMode(SWITCH_PIN, INPUT_PULLUP);
  pinMode(ZXLED_PIN, OUTPUT);
  ledcAttach(BUZZER_PIN, 2500, 8);

  oled.begin();
  Serial.println("กด ZX-SWITCH01 ค้างไว้เพื่อดูค่า mV ของ ZX-SOIL สำหรับ calibrate");
}

  int sMv = readMv(SOIL_PIN);
  float tempC = mcp9701C(tMv);
  float soil = soilPercent(sMv);

  bool alarm = tempC > TEMP_ALARM_C;

  digitalWrite(ZXLED_PIN, alarm ? HIGH : LOW);
  ledcWrite(BUZZER_PIN, alarm && (millis() / 500) % 2 ? 128 : 0);

  oled.setFont(u8g2_font_logisoso16_tr);
  oled.setCursor(0, 36);

  else oled.printf("%.1f C", tempC);

6I2C: ต่อเซ็นเซอร์ที่ไม่เคยเห็นภายใน 30 นาที

ผู้จัดจะให้เซ็นเซอร์ที่ยังไม่ระบุรุ่น บทนี้ฝึก วิธีคิด ไม่ใช่ท่องไลบรารี ใช้ SHT3x (อุณหภูมิ + ความชื้นสัมพัทธ์) เป็นตัวอย่าง เพราะเกี่ยวกับการตากและอบกาแฟโดยตรง

เรียนอะไร

  • I2C ใช้สาย 2 เส้น (SDA ข้อมูล, SCL สัญญาณนาฬิกา) ต่ออุปกรณ์หลายตัวบนบัสเดียว แยกกันด้วย แอดเดรส 7 บิต
  • บนบอร์ดนี้ OLED (0x3C) อยู่บนบัสเดียวกับพอร์ต K2 แล้ว จึงต่อเซ็นเซอร์เพิ่มได้เลยถ้าแอดเดรสไม่ซ้ำ
  • รูปแบบการคุยกับเซ็นเซอร์ I2C เกือบทุกตัว:
    1. beginTransmission(addr) → write(คำสั่ง/register) → endTransmission()
    2. รอเวลาวัดตาม datasheet
    3. requestFrom(addr, n) → read() n ไบต์
    4. รวมไบต์ ตรวจ CRC (ถ้ามี) แล้วแปลงเป็นหน่วยจริงด้วยสูตรใน datasheet
  • Wire.endTransmission() คืน 0 เมื่อมีอุปกรณ์ตอบ (ACK) — I2C scanner ใช้หลักนี้

สูตรต่อเซ็นเซอร์ใหม่ใน 30 นาที

นาทีทำอะไร
0–5อ่านชื่อชิปบนบอร์ด เปิด datasheet (เตรียมไว้ในเครื่องก่อนไป) หา: ไฟเลี้ยง (3.3 V ได้ไหม), interface, แอดเดรส
5–10ต่อสาย (VCC→3.3V, GND, SDA→21, SCL→22) แล้วรัน I2C scanner — ต้องเห็นแอดเดรสก่อนทำอย่างอื่น
10–20มีไลบรารีในเครื่องไหม? ถ้ามี เปิด example ของไลบรารีแล้วแก้ขา ถ้าไม่มี ทำตามตาราง "คำสั่งวัด" และ "สูตรแปลงค่า" ใน datasheet แบบในสเก็ตช์นี้
20–25ตรวจค่าให้สมเหตุสมผล (อุณหภูมิห้อง ~25–35 °C, RH 40–80%) ลองเป่าลม/จับเซ็นเซอร์ดูว่าค่าเปลี่ยน
25–30ห่อเป็นฟังก์ชัน bool readXxx(float &...) ที่คืน false เมื่ออ่านไม่ได้ แล้วนำไปรวมกับ firmware template (บทที่ 9)

ถ้า scanner ไม่เจออะไร: สลับ SDA/SCL, ตรวจไฟเลี้ยง, ตรวจว่าโมดูลมีตัวต้านทาน pull-up หรือไม่ (บนบอร์ด IPST-WiFi มี pull-up 1.8k ให้แล้ว)

ลองทำ

ต่อ SHT30/SHT31 ที่พอร์ต K2 แล้วอัปโหลด L06_i2c_sensor.ino ถ้าไม่มี SHT3x ให้รันเพื่อดูผลสแกน (จะเห็น OLED ที่ 0x3C) แล้วทำแบบฝึกหัดกับเซ็นเซอร์ I2C ที่มี

แบบฝึกหัด

  1. ทำให้สเก็ตช์ หาแอดเดรสเอง: ลอง 0x44 ถ้าไม่ตอบให้ลอง 0x45
  2. ใช้ คำสั่ง heater ของ SHT3x (ดู datasheet หัวข้อ Heater) เปิด 10 วินาทีแล้วสังเกตอุณหภูมิที่เพิ่ม — ทำไมเซ็นเซอร์ความชื้นต้องมี heater
  3. เปลี่ยนเป็นเซ็นเซอร์อื่นที่มีในห้อง (เช่น BH1750 ที่ 0x23: ส่งคำสั่ง 0x10 แล้วอ่าน 2 ไบต์ ÷ 1.2 = lux) โดยดู datasheet เท่านั้น จับเวลาว่าใช้กี่นาที
  4. ฝึกแบบวันแข่ง: ให้เพื่อนเลือกเซ็นเซอร์ 1 ตัวที่คุณไม่เคยใช้ (I2C, 1-Wire DS18B20, หรือ HX711 load cell) แล้วจับเวลาจนส่งค่าออก Serial ได้ — เป้าหมาย ≤ 30 นาที

// L06 · I2C: สแกนบัส แล้วอ่าน SHT3x (อุณหภูมิ/ความชื้นสัมพัทธ์) จาก datasheet โดยไม่ใช้ไลบรารี
// ต่อ SHT30/SHT31/SHT35 ที่พอร์ต K2 (I2C) — ใช้บัสเดียวกับ OLED

#include <Wire.h>

const int SDA_PIN = 21;
const int SCL_PIN = 22;
const uint8_t SHT3X_ADDR = 0x44;  // 0x45 ถ้าขา ADDR ต่อ VDD

void scanI2C() {
  Serial.println("สแกน I2C:");
  for (uint8_t addr = 1; addr < 127; addr++) {
    Wire.beginTransmission(addr);
    if (Wire.endTransmission() == 0) Serial.printf("  0x%02X\n", addr);
  }
}

// CRC-8 ตาม datasheet SHT3x: polynomial 0x31, ค่าเริ่ม 0xFF
uint8_t crc8(const uint8_t *data, int len) {
  uint8_t crc = 0xFF;
  for (int i = 0; i < len; i++) {
    crc ^= data[i];
    for (int b = 0; b < 8; b++) crc = (crc & 0x80) ? (crc << 1) ^ 0x31 : crc << 1;
  }
  return crc;
}

// วัดแบบ single shot, high repeatability, ไม่ใช้ clock stretching (คำสั่ง 0x2400)
bool readSht3x(float &tempC, float &rh) {
  Wire.beginTransmission(SHT3X_ADDR);
  Wire.write(0x24);
  Wire.write(0x00);
  if (Wire.endTransmission() != 0) return false;  // ไม่มีอุปกรณ์ตอบ

  delay(16);  // datasheet: high repeatability ใช้เวลาสูงสุด 15.5 ms

  if (Wire.requestFrom(SHT3X_ADDR, (uint8_t)6) != 6) return false;
  uint8_t d[6];
  for (int i = 0; i < 6; i++) d[i] = Wire.read();
  if (crc8(d, 2) != d[2] || crc8(d + 3, 2) != d[5]) return false;  // ข้อมูลเสีย

  uint16_t rawT = (d[0] << 8) | d[1];
  uint16_t rawRH = (d[3] << 8) | d[4];
  tempC = -45 + 175.0 * rawT / 65535.0;
  rh = 100.0 * rawRH / 65535.0;
  return true;
}

void setup() {
  Serial.begin(115200);
  delay(500);
  Wire.begin(SDA_PIN, SCL_PIN);
  Wire.setClock(100000);
  scanI2C();
}

void loop() {
  static unsigned long last = 0;
  if (millis() - last < 2000) return;
  last = millis();

  float t, rh;
  if (readSht3x(t, rh)) {
    Serial.printf("temp:%.2f,rh:%.1f\n", t, rh);
  } else {
    Serial.println("อ่าน SHT3x ไม่ได้ — ตรวจสาย/แอดเดรส");
  }
}

7Wi-Fi และเวลาจาก NTP

เรียนอะไร

  • WiFi.begin() ไม่รอ จนเชื่อมต่อสำเร็จ — ตัวอย่างทั่วไปมักมี while (WiFi.status() != WL_CONNECTED) delay(500); ซึ่งทำให้บอร์ดค้างทั้งตัวถ้า router ล่ม ในบทนี้ loop() ทำงานต่อได้เสมอ และ maintainWifi() คอยสั่งเชื่อมใหม่ทุก 10 วินาทีเมื่อหลุด
  • ESP32 ใช้ Wi-Fi 2.4 GHz เท่านั้น — ถ้า travel router ตั้งชื่อเดียวกันทั้ง 2.4/5 GHz ให้แยกชื่อ
  • NTP: configTzTime() ตั้งเขตเวลาและเซิร์ฟเวอร์ครั้งเดียว ระบบ SNTP จะ sync ให้เองในพื้นหลัง ใช้ time(nullptr) อ่านเวลา
  • ก่อน sync นาฬิกาเริ่มที่ปี 1970 — ห้ามส่งเวลานั้นออกไปเป็น ts ใช้ timeValid() ตรวจก่อนเสมอ
  • รูปแบบเวลาที่ตกลงกันในทีม: ISO 8601 พร้อม offset เช่น 2026-11-17T14:05:09+07:00 (Python อ่านได้ด้วย datetime.fromisoformat())
  • ในสนามแข่งอาจไม่มีอินเทอร์เน็ต: ให้ Pi หรือ notebook ของทีมเป็น NTP server (เช่น chrony ที่ตั้ง local stratum 10 และ allow 192.168.0.0/16) แล้วใส่ IP นั้นใน NTP_SERVER

ลองทำ

  1. แก้ WIFI_SSID และ WIFI_PASS ใน L07_wifi_ntp.ino แล้วอัปโหลด
  2. จอแสดง IP, ความแรงสัญญาณ (RSSI) และนาฬิกา — LED บนบอร์ดกระพริบเร็วระหว่างยังไม่เชื่อมต่อ
  3. ทดสอบความทน: ปิด hotspot/router 30 วินาทีแล้วเปิดใหม่ — บอร์ดต้องกลับมาเชื่อมต่อเองโดยไม่ต้องกด RESET และนาฬิกายังเดินต่อระหว่างหลุด
อย่า commit รหัส Wi-Fi จริงขึ้น Git — แก้ในเครื่องก่อนอัปโหลด หรือย้ายไปไว้ในไฟล์ secrets.h ที่อยู่ใน .gitignore

แบบฝึกหัด

  1. ใช้ WiFi.onEvent() พิมพ์ข้อความทุกครั้งที่ได้ IP และทุกครั้งที่หลุด พร้อมเหตุผล (info.wifi_sta_disconnected.reason)
  2. รองรับ Wi-Fi 2 ชุด (travel router ของทีม และ Wi-Fi สนาม) ด้วย WiFiMulti
  3. ตั้ง IP คงที่ ด้วย WiFi.config() — ข้อดีข้อเสียเทียบกับ DHCP ในวันแข่งคืออะไร
  4. ตั้ง NTP server บน Pi หรือ notebook แล้วทดสอบโดยตัดอินเทอร์เน็ตออก

// เวลาถูกต้องเมื่อปีมากกว่า 2024 (ก่อน sync นาฬิกาจะเริ่มที่ปี 1970)
bool timeValid() {
  return time(nullptr) > 1704067200;  // 2024-01-01T00:00:00Z
}

// เวลาท้องถิ่นแบบ ISO 8601 เช่น 2026-11-17T14:05:09+07:00
String isoTime() {
  if (!timeValid()) return "";
  time_t now = time(nullptr);
  struct tm tm;
  localtime_r(&now, &tm);
  char buf[32];
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%S%z", &tm);  // +0700
  String s(buf);
  s = s.substring(0, s.length() - 2) + ":" + s.substring(s.length() - 2);  // → +07:00
  return s;
}

8MQTT + JSON

บทนี้คือ "ESP32 → Mosquitto บน Pi ส่งข้อมูลได้" ซึ่งเป็นเป้าหมายของ ส. 24 ต.ค. ในแผน

เรียนอะไร

  • MQTT = publish/subscribe ผ่าน broker (Mosquitto บน Pi) ผู้ส่งไม่ต้องรู้ว่าใครรับ
  • Topic ที่ใช้ในบทนี้ (ปรับให้ตรงกับ Data Contract v1 ของทีม):
    Topicทิศทางเนื้อหา
    tgr2026/<team>/<node_id>/telemetryESP32 → brokerJSON ค่าเซ็นเซอร์ ทุก 5 วินาที
    tgr2026/<team>/<node_id>/statusESP32 → brokeronline / offline (retain)
    tgr2026/<team>/<node_id>/cmdbroker → ESP32JSON คำสั่ง
  • ทุกข้อความต้องมี lot_id + ts (หลักใน C1 ของแผน) — lot_id เชื่อมข้อมูลจาก sensor, vision และตลาดเข้าด้วยกัน ถ้ายังไม่ sync เวลา ให้ส่ง ts: null ไม่ใช่ปี 1970
  • ใส่ seq (เลขลำดับ) และ uptime_ms ด้วย ฝั่งรับจะรู้ว่า ข้อความหาย หรือ บอร์ดรีเซ็ต (seq กลับเป็น 0)
  • LWT (Last Will and Testament): บอกล่วงหน้าตอน connect ว่า "ถ้าฉันหายไปเฉย ๆ ให้ประกาศ offline ที่ topic status" — dashboard จึงรู้ว่า node ไหนตาย
  • Retain: broker เก็บข้อความล่าสุดของ topic ไว้ ผู้ที่ subscribe ทีหลังได้รับทันที (เหมาะกับ status ไม่เหมาะกับ telemetry)
  • QoS: PubSubClient publish ได้แค่ QoS 0 แต่ subscribe เป็น QoS 1 ได้ ถ้าต้องการ publish QoS 1 ให้ใช้ไลบรารี esp-mqtt ของ ESP-IDF (#include "mqtt_client.h")
  • ArduinoJson 7: ใช้ JsonDocument ตัวเดียว (ไม่ต้องกำหนดขนาดแบบ v6), doc["key"].is<T>() ตรวจชนิดก่อนอ่าน — คำสั่งที่ผิดรูปแบบต้องไม่ทำให้บอร์ดค้าง
  • mqtt.setBufferSize(512) — ค่าเริ่มต้น 256 ไบต์ ข้อความ JSON ที่ยาวกว่านั้นจะ publish ไม่ออกโดยไม่มี error ให้เห็น

String isoTimeUtc() {

  struct tm tm;
  gmtime_r(&now, &tm);

  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", &tm);

void onMessage(char *topic, byte *payload, unsigned int len) {
  JsonDocument cmd;
  DeserializationError err = deserializeJson(cmd, payload, len);

    Serial.printf("cmd JSON ผิดรูปแบบ: %s\n", err.c_str());
    return;

  Serial.print("cmd: ");
  serializeJson(cmd, Serial);
  Serial.println();

  if (cmd["led"].is<bool>()) digitalWrite(LED_PIN, cmd["led"].as<bool>() ? HIGH : LOW);

สิ่งใหม่ในบทนี้

  • State machine — ทุกการตัดสินใจขึ้นกับสถานะปัจจุบัน และทุกครั้งที่เปลี่ยนสถานะจะส่ง event ให้ service บันทึกเป็นประวัติของล็อต (traceability)
  • Hysteresis — เปิดฮีตเตอร์เมื่อต่ำกว่า SP − band ปิดเมื่อสูงกว่า SP + band ระหว่างนั้นคงสถานะเดิม รีเลย์จึงไม่สลับถี่จนพัง
  • Fail-safe — ฮีตเตอร์ปิดตอนบูตเสมอ, เซ็นเซอร์อ่านค่าผิดปกติ = fault, เปลี่ยน lot_id ระหว่างอบไม่ได้
  • Watchdog — ถ้า loop() ค้างเกิน 15 วินาที บอร์ดรีเซ็ตตัวเองและกลับสู่สถานะปลอดภัย
  • ตั้งค่าแยกไฟล์ — แก้ config.h ไฟล์เดียว ไม่ต้องแตะโค้ดควบคุม
  • Topic event แยกจาก telemetry — ข้อมูลต่อเนื่องกับเหตุการณ์สำคัญมีการใช้งานต่างกัน

int readMv(int pin) {
  const int N = 9;
  int v[N];
  for (int i = 0; i < N; i++) v[i] = analogReadMilliVolts(pin);
  for (int i = 1; i < N; i++) {
    int x = v[i], j = i - 1;
    while (j >= 0 && v[j] > x) { v[j + 1] = v[j]; j--; }
    v[j + 1] = x;
  }
  return v[N / 2];
}

void readSensors() {
  static float emaT = NAN, emaM = NAN;
  tempMv = readMv(TEMP_PIN);
  moistMv = readMv(MOIST_PIN);
  float t = (tempMv - 400) / 19.5;
  float m = constrain(100.0 * (moistMv - MOIST_DRY_MV) / (MOIST_WET_MV - MOIST_DRY_MV), 0, 100);

  tempC = emaT;
  moistPct = emaM;
}

bool timeValid() { return time(nullptr) > 1704067200; }

String isoTimeUtc() {
  time_t now = time(nullptr);
  struct tm tm;
  gmtime_r(&now, &tm);
  char buf[25];
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", &tm);
  return buf;
}

void addHeader(JsonDocument &doc) {
  doc["lot_id"] = lotId;
  doc["node_id"] = nodeId;
  if (timeValid()) doc["ts"] = isoTimeUtc();
  else doc["ts"] = nullptr;
  doc["seq"] = seq++;
}

void publishJson(const String &topic, JsonDocument &doc) {
  char buf[512];
  size_t n = serializeJson(doc, buf);
  mqtt.publish(topic.c_str(), buf, n);
  Serial.println(buf);
}

void publishTelemetry() {
  if (!mqtt.connected()) return;
  JsonDocument doc;
  addHeader(doc);
  doc["state"] = STATE_NAME[state];
  doc["temp_c"] = round(tempC * 10) / 10.0;
  doc["moist_pct"] = round(moistPct * 10) / 10.0;
  doc["setpoint_c"] = setpointC;
  doc["target_pct"] = targetPct;
  doc["heater"] = heaterOn;
  doc["fan_duty"] = fanDuty;
  doc["rssi"] = WiFi.RSSI();
  doc["uptime_ms"] = millis();
  publishJson(topicTelemetry, doc);
}

// เหตุการณ์สำคัญ (เปลี่ยนสถานะ, fault) ส่งแยก topic เพื่อให้ service บันทึกเป็นประวัติของล็อต
void publishEvent(const char *event, const String &detail = "") {
  if (!mqtt.connected()) return;
  JsonDocument doc;
  addHeader(doc);
  doc["event"] = event;
  doc["state"] = STATE_NAME[state];
  if (detail.length()) doc["detail"] = detail;
  publishJson(topicEvent, doc);
}

void beep(int ms) {
  ledcWrite(BUZZER_PIN, 128);
  delay(ms);  // สั้นมาก ยอมให้บล็อกได้
  ledcWrite(BUZZER_PIN, 0);
}

void setState(State s, const String &detail = "") {
  if (s == state) return;
  state = s;
  if (s == FAULT) faultReason = detail;
  Serial.printf("state → %s %s\n", STATE_NAME[s], detail.c_str());
  publishEvent("state", detail);
  lastPublish = 0;  // ส่ง telemetry ทันที
  beep(s == FAULT ? 400 : 80);
}

void control() {
  if (state == DRYING) {
    if (!tempSensorOk()) setState(FAULT, "temp sensor");
    else if (tempC > TEMP_MAX_C) setState(FAULT, "over temp");
    else if (moistPct <= targetPct) setState(DONE, "target reached");
  }

  if (state == DRYING) {
    // hysteresis: อยู่ในแถบ ±band ให้คงสถานะเดิม ฮีตเตอร์จึงไม่สลับถี่
    if (tempC < setpointC - bandC) heaterOn = true;
    else if (tempC > setpointC + bandC) heaterOn = false;
    // พัดลม: หมุนเบาตลอด และเร่งเมื่อร้อนเกินเป้า (proportional)
    float over = tempC - setpointC;
    fanDuty = constrain(FAN_MIN_DUTY + (int)(over * 40), FAN_MIN_DUTY, 255);
  } else {
    heaterOn = false;
    fanDuty = (state == FAULT && tempC > TEMP_MAX_C) ? 255 : 0;  // ร้อนเกิน → เป่าระบาย
  }

  digitalWrite(HEATER_PIN, heaterOn ? HIGH : LOW);
  ledcWrite(FAN_PIN, fanDuty);
}

void startDrying() {
  if (state == FAULT) return;
  setState(DRYING, "start");
}

void stopDrying() {
  setState(IDLE, "stop");
}

void onMessage(char *topic, byte *payload, unsigned int len) {
  JsonDocument cmd;
  if (deserializeJson(cmd, payload, len)) {
    publishEvent("cmd_error", "invalid json");
    return;
  }
  if (cmd["lot_id"].is<const char *>()) {
    if (state == DRYING) {
      publishEvent("cmd_error", "stop before changing lot_id");
    } else {
      lotId = cmd["lot_id"].as<const char *>();
      if (state == DONE) setState(IDLE, "new lot");
    }
  }
  if (cmd["setpoint_c"].is<float>()) setpointC = constrain(cmd["setpoint_c"].as<float>(), 25.0f, TEMP_MAX_C - 5);
  if (cmd["band_c"].is<float>()) bandC = constrain(cmd["band_c"].as<float>(), 0.5f, 10.0f);
  if (cmd["target_pct"].is<float>()) targetPct = constrain(cmd["target_pct"].as<float>(), 0.0f, 100.0f);
  if (cmd["start"] == true) startDrying();
  if (cmd["stop"] == true) stopDrying();
  if (cmd["reset"] == true && state == FAULT) setState(IDLE, "reset");
  lastPublish = 0;
}

void maintainNetwork() {
  if (millis() - lastWifiCheck > 10000) {
    lastWifiCheck = millis();
    if (WiFi.status() != WL_CONNECTED) {
      WiFi.disconnect();
      WiFi.begin(WIFI_SSID, WIFI_PASS);
    }
  }
  if (!mqtt.connected() && WiFi.status() == WL_CONNECTED && millis() - lastMqttTry > 5000) {
    lastMqttTry = millis();
    if (mqtt.connect(nodeId.c_str(), nullptr, nullptr, topicStatus.c_str(), 1, true, "offline")) {
      mqtt.publish(topicStatus.c_str(), "online", true);
      mqtt.subscribe(topicCmd.c_str(), 1);
      publishEvent("boot_or_reconnect");
    }
  }
  mqtt.loop();
}

void handleButton() {
  static bool last = HIGH;
  static unsigned long lastChange = 0;
  bool b = digitalRead(BUTTON_PIN);
  if (b != last && millis() - lastChange > 30) {
    lastChange = millis();
    last = b;
    if (b == LOW) {
      if (state == DRYING) stopDrying();
      else if (state == FAULT) setState(IDLE, "reset");
      else startDrying();
    }
  }
}

void drawScreen() {
  oled.clearBuffer();
  oled.setFont(u8g2_font_6x10_tf);

  oled.setCursor(0, 9);
  oled.printf("%-8s %s", lotId.c_str(), STATE_NAME[state]);
  oled.drawStr(98, 9, WiFi.status() == WL_CONNECTED ? (mqtt.connected() ? "MQTT" : "WiFi") : "----");
  oled.drawHLine(0, 12, 128);

  oled.setFont(u8g2_font_logisoso16_tr);
  oled.setCursor(0, 32);
  oled.printf("%.1fC", tempC);
  oled.setCursor(66, 32);
  oled.printf("%.0f%%", moistPct);

  oled.setFont(u8g2_font_6x10_tf);
  oled.setCursor(0, 44);
  oled.printf("SP %.0f+-%.0f", setpointC, bandC);
  oled.setCursor(66, 44);
  oled.printf("TG %.0f%%", targetPct);

  oled.setCursor(0, 58);
  if (state == FAULT) {
    oled.printf("FAULT: %s", faultReason.c_str());
  } else {
    oled.printf("HEAT %s", heaterOn ? "ON " : "off");
    oled.setCursor(66, 58);
    oled.printf("FAN %d%%", fanDuty * 100 / 255);
  }
  oled.sendBuffer();
}

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT);
  pinMode(HEATER_PIN, OUTPUT);
  digitalWrite(HEATER_PIN, LOW);  // ปลอดภัยไว้ก่อน: ฮีตเตอร์ปิดตอนบูต
  ledcAttach(FAN_PIN, 25000, 8);  // 25 kHz พ้นช่วงที่หูได้ยิน
  ledcAttach(BUZZER_PIN, 2500, 8);

  oled.setI2CAddress(0x3C * 2);
  oled.begin();

  WiFi.mode(WIFI_STA);
  WiFi.setAutoReconnect(true);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  configTzTime("<+07>-7", NTP_SERVER);

  uint8_t mac[6];
  WiFi.macAddress(mac);
  char id[16];
  snprintf(id, sizeof(id), "esp32-%02x%02x%02x", mac[3], mac[4], mac[5]);
  nodeId = id;
  String base = String("tgr2026/") + TEAM + "/" + nodeId;
  topicTelemetry = base + "/telemetry";
  topicStatus = base + "/status";
  topicEvent = base + "/event";
  topicCmd = base + "/cmd";
  Serial.printf("node_id = %s\n", nodeId.c_str());

  mqtt.setServer(MQTT_HOST, MQTT_PORT);
  mqtt.setCallback(onMessage);
  mqtt.setBufferSize(512);
  mqtt.setKeepAlive(15);

  // Watchdog: ถ้า loop() ค้างเกิน 15 วินาที ให้รีเซ็ตบอร์ด (ฮีตเตอร์จะปิดตอนบูต)
  esp_task_wdt_config_t wdt = {.timeout_ms = 15000, .idle_core_mask = 0, .trigger_panic = true};
  if (esp_task_wdt_reconfigure(&wdt) != ESP_OK) esp_task_wdt_init(&wdt);
  esp_task_wdt_add(NULL);

  readSensors();
}

void loop() {
  esp_task_wdt_reset();
  maintainNetwork();
  handleButton();

  static unsigned long lastControl = 0;
  if (millis() - lastControl >= 500) {
    lastControl = millis();
    readSensors();
    control();
    drawScreen();
    // LED: ติดค้าง = กำลังอบ, กระพริบ = fault, ดับ = idle/done
    digitalWrite(LED_PIN, state == DRYING || (state == FAULT && (millis() / 500) % 2));
  }

  if (lastPublish == 0 || millis() - lastPublish >= PUBLISH_MS) {
    lastPublish = millis();
    publishTelemetry();
  }
}
